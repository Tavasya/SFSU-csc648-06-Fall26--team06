/**
 * Public interfaces supporting R1.3, R2.1, R2.2.
 * @module components/gatoraid/EventSearch
 */
"use client";

import { getScheduleAvailability } from "@/lib/schedule/conflicts";
import { useMemo, useState } from "react";
import Link from "next/link";
import {
  CalendarDays,
  CalendarX2,
  CircleCheck,
  CircleHelp,
  Info,
  MapPin,
  Search,
  X,
  ChevronDown,
} from "lucide-react";
import {
  bannerFor,
  bannerStyle,
  orderedCategoryNames,
} from "@/lib/events/banner";
import { SORT_OPTIONS, sortEvents, sortNote } from "@/lib/events/sort";
import {
  availabilityFor,
  eventLocations,
  filterEvents,
  hasActiveFilters,
} from "@/lib/events/search";

/**
 * Render searchable, filterable event cards with sorting, schedule-fit and host visibility controls.
 * @function EventSearch
 * @memberof module:components/gatoraid/EventSearch
 * @static
 * @param {Object} props - Component properties.
 * @param {*} props.events - Event records to display.
 * @param {*} props.categories - Available event categories.
 * @param {*} [props.schedule] - Saved commitments used to check availability.
 * @param {*} [props.scheduleStatus] - The schedule status setting or callback.
 * @param {*} [props.signedIn] - Whether the current viewer is authenticated.
 * @param {*} [props.interestIds] - Selected interest category IDs.
 * @returns {React.ReactElement} Rendered component tree.
 */
export default function EventSearch({
  events,
  categories,
  schedule = [],
  scheduleStatus = "unused",
  signedIn = false, // shows the "Show my events" chip (events marked hosted_by_you)
  interestIds = null, // For you: the student's interests, for the "Best match" sort
}) {
  const [query, setQuery] = useState("");
  const [category, setCategory] = useState("");
  const [date, setDate] = useState("");
  const [location, setLocation] = useState("");
  const [access, setAccess] = useState(""); // "", "wheelchair", "asl" or "both"
  const [scheduleFilter, setScheduleFilter] = useState("all");
  // Events the signed-in user hosts are hidden unless they turn this on.
  const [showMine, setShowMine] = useState(false);
  const [sort, setSort] = useState(interestIds?.length ? "match" : "soonest");
  const myEventCount = events.filter((event) => event.hosted_by_you).length;

  const locations = useMemo(() => eventLocations(events), [events]);

  const filteredEvents = useMemo(
    () =>
      filterEvents(
        events,
        categories,
        { query, category, date, location, access, scheduleFilter, showMine },
        { schedule, scheduleStatus }
      ),
    [
      events,
      categories,
      query,
      category,
      date,
      location,
      access,
      schedule,
      scheduleStatus,
      scheduleFilter,
      showMine,
    ]
  );

  const hasSchedule = scheduleStatus === "ready" && schedule.length > 0;
  const sortOptions = SORT_OPTIONS.filter(
    (option) =>
      !option.needs ||
      (option.needs === "interests" && interestIds?.length) ||
      (option.needs === "schedule" && hasSchedule)
  );
  const activeSort = sortOptions.some((option) => option.id === sort) ? sort : "soonest";
  const { sortedEvents, note } = useMemo(() => {
    const options = {
      interestIds: interestIds ?? [],
      fits: (event) =>
        hasSchedule &&
        getScheduleAvailability(event, schedule).status === "available",
    };
    return {
      sortedEvents: sortEvents(filteredEvents, activeSort, options),
      note: sortNote(filteredEvents, activeSort, options),
    };
  }, [filteredEvents, activeSort, interestIds, hasSchedule, schedule]);

  function clearFilters() {
    setQuery("");
    setCategory("");
    setDate("");
    setLocation("");
    setAccess("");
    setScheduleFilter("all");
  }

  const hasFilters = hasActiveFilters({
    query,
    category,
    date,
    location,
    access,
    scheduleFilter,
  });

  return (
    <section
      className="space-y-5"
      aria-label="Search campus events"
    >
      <div className="grid grid-cols-1 gap-3 md:grid-cols-2 lg:grid-cols-5">

        {/* Search bar */}
        <div className="relative md:col-span-2 lg:col-span-5">
          <Search
            size={19}
            className="absolute left-3 top-1/2 -translate-y-1/2 ga-muted"
          />

          <input
            type="search"
            value={query}
            onChange={(event) => setQuery(event.target.value)}
            placeholder="Search events, topics, or keywords..."
            aria-label="Search events by keyword"
            style={{ paddingLeft: "48px" }}
            className="w-full rounded-lg border border-gray-300 bg-white py-3 pr-3 text-sm text-gray-900"
          />
        </div>

        {/* Category filter */}
        <div className="relative min-w-0">
          <select
            value={category}
            onChange={(event) => setCategory(event.target.value)}
            aria-label="Filter by category"
            className="w-full appearance-none rounded-lg border border-gray-300 bg-white py-3 pl-3 pr-10 text-sm text-gray-900"
          >
            <option value="">All categories</option>

            {categories.map((item) => (
              <option key={item.id} value={item.name}>
                {item.name}
              </option>
            ))}
          </select>

          <ChevronDown
            size={16}
            aria-hidden="true"
            className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-gray-600"
          />
        </div>

        {/* Date filter */}
        <input
          type="date"
          value={date}
          onChange={(event) => setDate(event.target.value)}
          aria-label="Filter by event date"
          className="w-full min-w-0 rounded-lg border border-gray-300 bg-white p-3 text-sm text-gray-900"
        />

        {/* Location filter */}
        <div className="relative min-w-0">
          <select
            value={location}
            onChange={(event) => setLocation(event.target.value)}
            aria-label="Filter by location"
            className="w-full appearance-none rounded-lg border border-gray-300 bg-white py-3 pl-3 pr-10 text-sm text-gray-900"
          >
            <option value="">All locations</option>

            {locations.map((name) => (
              <option key={name} value={name}>
                {name}
              </option>
            ))}
          </select>

          <ChevronDown
            size={16}
            aria-hidden="true"
            className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-gray-600"
          />
        </div>

        {/* Accessibility filter */}
        <div className="relative min-w-0">
          <select
            value={access}
            onChange={(event) => setAccess(event.target.value)}
            aria-label="Filter by accessibility"
            className="w-full appearance-none rounded-lg border border-gray-300 bg-white py-3 pl-3 pr-10 text-sm text-gray-900"
          >
            <option value="">Any accessibility</option>
            <option value="wheelchair">Wheelchair accessible</option>
            <option value="asl">ASL interpretation</option>
            <option value="both">Wheelchair and ASL</option>
          </select>

          <ChevronDown
            size={16}
            aria-hidden="true"
            className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-gray-600"
          />
        </div>

        {/* Clear filters */}
        <button
          type="button"
          onClick={clearFilters}
          disabled={!hasFilters}
          className="inline-flex items-center justify-center gap-2 rounded-lg border border-gray-300 px-3 py-2 text-sm disabled:cursor-not-allowed disabled:opacity-50"
        >
          <X size={16} />
          Clear filters
        </button>
      </div>


      {/* Toggles: schedule availability and (signed in) your own events */}
      {(scheduleStatus !== "unused" || signedIn) && (
        <div className="flex flex-wrap items-center gap-x-8 gap-y-3">
          {scheduleStatus !== "unused" && (
            <div className="flex flex-wrap items-center gap-3">
              <span className="text-sm font-medium text-[#292437]">
                Schedule availability:
              </span>

              <div className="inline-flex rounded-full border border-[#D8CDE9] bg-white p-1">
                <button
                  type="button"
                  onClick={() => setScheduleFilter("all")}
                  aria-pressed={scheduleFilter === "all"}
                  className={`rounded-full px-4 py-2 text-sm font-medium transition ${scheduleFilter === "all"
                    ? "bg-[#65449B] text-white"
                    : "text-[#65449B] hover:bg-[#F5F1FA]"
                    }`}
                >
                  All events
                </button>

                <button
                  type="button"
                  onClick={() => setScheduleFilter("available")}
                  disabled={scheduleStatus !== "ready" || schedule.length === 0}
                  aria-pressed={scheduleFilter === "available"}
                  className={`rounded-full px-4 py-2 text-sm font-medium transition disabled:cursor-not-allowed disabled:opacity-40 ${scheduleFilter === "available"
                    ? "bg-[#65449B] text-white"
                    : "text-[#65449B] hover:bg-[#F5F1FA]"
                    }`}
                >
                  Fits my schedule
                </button>
              </div>

              {scheduleStatus === "loading" && (
                <span className="text-xs text-gray-500">
                  Checking your schedule...
                </span>
              )}

              {scheduleStatus === "ready" && schedule.length === 0 && (
                <Link
                  href="/schedule"
                  className="text-sm font-medium text-[#65449B] hover:underline"
                >
                  Add your schedule
                </Link>
              )}
            </div>
          )}

          {/* Events the user hosts are hidden unless they show them */}
          {signedIn && (
            <div
              className="flex flex-wrap items-center gap-3"
              title={
                myEventCount
                  ? "Include the events you host, to see how they look to students"
                  : "You don't host any upcoming published events"
              }
            >
              <span className="text-sm font-medium text-[#292437]">
                Your events:
              </span>

              <div className="inline-flex rounded-full border border-[#D8CDE9] bg-white p-1">
                <button
                  type="button"
                  onClick={() => setShowMine(false)}
                  aria-pressed={!showMine}
                  className={`rounded-full px-4 py-2 text-sm font-medium transition ${!showMine
                    ? "bg-[#65449B] text-white"
                    : "text-[#65449B] hover:bg-[#F5F1FA]"
                    }`}
                >
                  Hide
                </button>

                <button
                  type="button"
                  onClick={() => setShowMine(true)}
                  disabled={myEventCount === 0}
                  aria-pressed={showMine}
                  className={`rounded-full px-4 py-2 text-sm font-medium transition disabled:cursor-not-allowed disabled:opacity-40 ${showMine
                    ? "bg-[#65449B] text-white"
                    : "text-[#65449B] hover:bg-[#F5F1FA]"
                    }`}
                >
                  Show my events ({myEventCount})
                </button>
              </div>
            </div>
          )}
        </div>
      )}

      {/* Results count and sort order */}
      <div className="flex flex-wrap items-center justify-between gap-3">
        <p className="text-sm ga-muted" aria-live="polite">
          {filteredEvents.length}{" "}
          {filteredEvents.length === 1 ? "event" : "events"} found
        </p>

        <div className="flex items-center gap-2">
          <label htmlFor="event-sort" className="text-sm font-medium text-[#292437]">
            Sort by
          </label>
          <div className="relative">
            <select
              id="event-sort"
              value={activeSort}
              onChange={(event) => setSort(event.target.value)}
              style={{ paddingRight: "36px" }}
              className="appearance-none rounded-lg border border-gray-300 bg-white py-2 pl-3 text-sm text-gray-900"
            >
              {sortOptions.map((option) => (
                <option key={option.id} value={option.id}>
                  {option.label}
                </option>
              ))}
            </select>
            <ChevronDown
              size={15}
              aria-hidden="true"
              className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-gray-600"
            />
          </div>
        </div>
      </div>

      {note && (
        <p role="status" className="flex items-start gap-2 text-sm ga-muted">
          <Info size={15} className="mt-0.5 shrink-0" aria-hidden="true" />
          {note}
        </p>
      )}

      {/* Event results */}
      {filteredEvents.length > 0 ? (
        <div className="ga-public-grid">
          {sortedEvents.map((event) => {
            const categoryNames = orderedCategoryNames(
              event.category_ids,
              categories
            );

            const banner = bannerFor(categoryNames[0]);
            const availability = availabilityFor(
              event,
              schedule,
              scheduleStatus
            );

            return (
              <Link
                href={`/events/${event.id}`}
                className="ga-card ga-public-card"
                key={event.id}
              >
                <div
                  className="ga-cover"
                  style={bannerStyle(banner)}
                >
                  <banner.Icon size={45} />
                </div>

                <div className="ga-chip-group ga-public-card-chips">
                  {/* Your plans page: Going, Interested, Saved, Not going */}
                  {event.hosted_by_you && <span className="ga-badge plan">You&apos;re hosting</span>}
                  {(event.plan_labels ?? []).map((label) => (
                    <span className="ga-badge plan" key={label}>
                      {label}
                    </span>
                  ))}
                  {categoryNames.map((name) => (
                    <span className="ga-badge" key={name}>
                      {name}
                    </span>
                  ))}
                </div>

                <h2>{event.title}</h2>

                {scheduleStatus !== "unused" && (
                  <div className="ga-fit-row">
                    {availability.status === "available" && (
                      <span className="ga-fit fits">
                        <CircleCheck size={13} aria-hidden="true" />
                        Fits your schedule
                      </span>
                    )}

                    {availability.status === "conflict" && (
                      <>
                        <span className="ga-fit conflict">
                          <CalendarX2 size={13} aria-hidden="true" />
                          Schedule conflict
                        </span>
                        <div className="ga-fit-detail">
                          Overlaps with{" "}
                          {availability.conflicts
                            .map((course) => course.name || course.class_name)
                            .join(", ")}
                        </div>
                      </>
                    )}

                    {availability.status === "unknown" && (
                      <span className="ga-fit unknown">
                        <CircleHelp size={13} aria-hidden="true" />
                        Schedule fit unknown
                      </span>
                    )}
                  </div>
                )}

                {event.recommendation_reasons?.length > 0 && (
                  <p className="ga-muted student-match">
                    Matches your interests: {event.recommendation_reasons.join(", ")}
                  </p>
                )}
                {event.status === "cancelled" && (
                  <div className="ga-cancelled-notice">
                    <span className="ga-badge cancelled">Cancelled</span>

                    {event.cancellation_message && (
                      <p className="ga-cancellation-message">
                        {event.cancellation_message}
                      </p>
                    )}
                  </div>
                )}

                <p className="ga-muted">
                  {new Date(
                    event.start_date_time
                  ).toLocaleDateString("en-US", {
                    month: "long",
                    day: "numeric",
                    timeZone: "America/Los_Angeles",
                  })}
                </p>

                <p>
                  <MapPin size={15} />
                  {event.locations?.building_name}
                </p>

                <span className="ga-badge price-badge">
                  {event.admission_type === "free"
                    ? "Free"
                    : "Paid"}
                </span>
              </Link>
            );
          })}
        </div>
      ) : (
        /* No results message */
        <div className="ga-empty ga-card">
          <CalendarDays size={40} />


          <h2>
            {scheduleFilter === "available"
              ? "No events fit your schedule"
              : "No matching events found"}
          </h2>

          <p className="ga-muted">
            {scheduleFilter === "available"
              ? "No events match your current filters without a schedule conflict. Try All events to see more options."
              : "Try another keyword or adjust your filters."}
          </p>


          <button
            type="button"
            className="ga-button"
            onClick={clearFilters}
          >
            Clear filters
          </button>
        </div>
      )}
    </section>
  );
}
